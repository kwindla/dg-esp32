cmake_minimum_required(VERSION 3.22)
project(diagnostics_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenSSL REQUIRED)
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.18.0.tar.gz
    URL_HASH SHA256=6e3191c1455468b3fc35a417fb565c1c5071aee1b7e7f85e30cf48a98d37d8b5)
FetchContent_MakeAvailable(googletest)
enable_testing()
include(GoogleTest)
add_library(dump_check ../dump_check.cpp)
target_include_directories(dump_check PUBLIC ../include fakes)
target_link_libraries(dump_check PUBLIC OpenSSL::Crypto)
add_executable(dump_fixture dump_fixture.cpp)
target_link_libraries(dump_fixture PRIVATE dump_check)
foreach(target s3 p4 p4_rev3)
    set(sources test_diagnostics.cpp ../uart.cpp ../console_backend.cpp)
    if(target STREQUAL "s3")
        list(APPEND sources ../usb_serial_jtag.cpp)
    endif()
    add_executable(diagnostics_${target}_tests ${sources})
    target_include_directories(diagnostics_${target}_tests PRIVATE .. ../../dg_board/include ../../../main)
    target_compile_definitions(diagnostics_${target}_tests PRIVATE
        TEST_P4=$<NOT:$<STREQUAL:${target},s3>> TEST_P4_REV3=$<STREQUAL:${target},p4_rev3>)
    target_link_libraries(diagnostics_${target}_tests PRIVATE dump_check GTest::gtest_main)
    target_compile_options(diagnostics_${target}_tests PRIVATE -Wall -Wextra -Werror)
    gtest_discover_tests(diagnostics_${target}_tests TEST_PREFIX ${target}.)
endforeach()
add_test(NAME ConsoleChoice.RejectUsbOnP4 COMMAND ${CMAKE_COMMAND}
    -DCXX=${CMAKE_CXX_COMPILER} -DSOURCE=${CMAKE_CURRENT_SOURCE_DIR}
    -DOUTPUT=${CMAKE_CURRENT_BINARY_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/reject_console.cmake)

foreach(kind uart usb)
    add_executable(console_output_${kind}_test test_console_output.cpp ../console_output.cpp ../dump_check.cpp)
    target_include_directories(console_output_${kind}_test PRIVATE output_fakes fakes ../include .. ../../dg_board/include)
    target_compile_definitions(console_output_${kind}_test PRIVATE TEST_P4=$<STREQUAL:${kind},uart> TEST_P4_REV3=0)
    target_compile_options(console_output_${kind}_test PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(console_output_${kind}_test PRIVATE OpenSSL::Crypto GTest::gtest_main)
    gtest_discover_tests(console_output_${kind}_test TEST_PREFIX output_${kind}.)
endforeach()
