cmake_minimum_required(VERSION 3.22)
project(wifi_observe_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(googletest
 URL https://github.com/google/googletest/archive/refs/tags/v1.18.0.tar.gz
 URL_HASH SHA256=6e3191c1455468b3fc35a417fb565c1c5071aee1b7e7f85e30cf48a98d37d8b5)
set(INSTALL_GTEST OFF)
set(BUILD_GMOCK OFF)
FetchContent_MakeAvailable(googletest)
enable_testing()
include(GoogleTest)
foreach(mode enabled disabled)
 add_executable(wifi_observe_${mode} test_wifi_observe.cpp)
 target_include_directories(wifi_observe_${mode} PRIVATE fakes ../include ../../dg_board/test_host/include ../../dg_board/include)
 target_link_libraries(wifi_observe_${mode} PRIVATE GTest::gtest_main)
 target_compile_options(wifi_observe_${mode} PRIVATE -Wall -Wextra -Werror)
 if(mode STREQUAL enabled)
  target_compile_definitions(wifi_observe_${mode} PRIVATE CONFIG_DG_WIFI_OBSERVE=1)
 endif()
 gtest_discover_tests(wifi_observe_${mode})
endforeach()
# Native Wi-Fi observation is S3-only. Both S3 boards exercise it; the P4
# selector checks only the disabled body, which must have no driver symbols.
foreach(board waveshare146 p4xc)
 foreach(mode enabled disabled)
  if(board STREQUAL p4xc AND mode STREQUAL enabled)
   continue()
  endif()
  add_executable(wifi_observe_${board}_${mode} test_wifi_observe.cpp)
  target_include_directories(wifi_observe_${board}_${mode} PRIVATE fakes ../include ../../dg_board/test_host/include ../../dg_board/include)
  if(board STREQUAL p4xc)
   target_compile_definitions(wifi_observe_${board}_${mode} PRIVATE DG_TEST_BOARD=3)
  else()
   target_compile_definitions(wifi_observe_${board}_${mode} PRIVATE DG_TEST_BOARD=2)
  endif()
  if(mode STREQUAL enabled)
   target_compile_definitions(wifi_observe_${board}_${mode} PRIVATE CONFIG_DG_WIFI_OBSERVE=1)
  endif()
  target_link_libraries(wifi_observe_${board}_${mode} PRIVATE GTest::gtest_main)
  target_compile_options(wifi_observe_${board}_${mode} PRIVATE -Wall -Wextra -Werror)
  gtest_discover_tests(wifi_observe_${board}_${mode} TEST_PREFIX ${board}.)
 endforeach()
endforeach()
