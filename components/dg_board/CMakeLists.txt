set(board_sources "src/common/gpio_isr.cpp" "src/common/touch.cpp")
unset(board_backend)
if(CONFIG_DG_BOARD_ECHOEAR)
    set(board_backend "echoear")
elseif(CONFIG_DG_BOARD_WAVESHARE_146)
    set(board_backend "waveshare146")
elseif(CONFIG_DG_BOARD_P4XC)
    set(board_backend "p4xc")
endif()
if(DEFINED board_backend)
    foreach(source board power info audio_hw)
        list(APPEND board_sources "src/${board_backend}/${source}.cpp")
    endforeach()
    if(CONFIG_DG_BOARD_P4XC)
        list(APPEND board_sources "src/p4xc/touch_gt911.cpp")
    else()
        list(APPEND board_sources "src/${board_backend}/touch.cpp")
    endif()
endif()
if(IDF_TARGET STREQUAL "esp32s3")
    list(APPEND board_sources "src/common/cache_s3.cpp")
else()
    list(APPEND board_sources "src/common/cache_null.cpp")
endif()
# Source selection only: dependency discovery must not depend on CONFIG values.
idf_component_register(SRCS ${board_sources} INCLUDE_DIRS "include" REQUIRES esp_driver_i2c freertos PRIV_REQUIRES esp_hw_support pthread esp_driver_gpio esp_driver_ledc esp_timer heap)
target_compile_features(${COMPONENT_LIB} PRIVATE cxx_std_17)
