cmake_minimum_required(VERSION 3.20)
project(display_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenSSL REQUIRED)
include(FetchContent)
FetchContent_Declare(nlohmann_json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(nlohmann_json)
add_library(display_core ../src/present/logical_adapter.cpp ../src/present/presenter.cpp ../backends/null/null_panel.cpp ../ui/simple/character.cpp ../src/dgi.cpp ../src/scene.cpp ../src/arena.cpp ../src/cache.cpp ../src/clock.cpp ../src/engine.cpp ../src/flash_store.cpp ../src/json_request.cpp)
target_link_libraries(display_core PUBLIC OpenSSL::Crypto nlohmann_json::nlohmann_json)
target_include_directories(display_core PUBLIC ../include ../../dg_board/include)
option(DG_BOARD_ECHOEAR "Test EchoEar 360px panel" OFF)
if(DG_BOARD_ECHOEAR)
    target_compile_definitions(display_core PUBLIC CONFIG_DG_BOARD_ECHOEAR=1)
endif()
target_compile_options(display_core PRIVATE -Wall -Wextra -Werror)
add_executable(dgi_cli dgi_cli.cpp)
target_link_libraries(dgi_cli PRIVATE display_core)
enable_testing()
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.18.0.tar.gz URL_HASH SHA256=6e3191c1455468b3fc35a417fb565c1c5071aee1b7e7f85e30cf48a98d37d8b5)
set(INSTALL_GTEST OFF)
set(BUILD_GMOCK OFF)
FetchContent_MakeAvailable(googletest)
add_executable(display_core_test core_test.cpp character_test.cpp standalone_test.cpp bounded_mutex_test.cpp present_test.cpp)
target_include_directories(display_core_test PRIVATE ../../dg_audio/include ../../dg_board/include)
target_link_libraries(display_core_test PRIVATE display_core GTest::gtest_main)
target_compile_definitions(display_core_test PRIVATE TEST_STORE_ROOT="${CMAKE_BINARY_DIR}")
include(GoogleTest)
gtest_discover_tests(display_core_test)

add_executable(touch_driver_test ../../dg_board/src/common/gpio_isr.cpp ../../dg_board/src/common/touch.cpp ../../dg_board/src/echoear/touch.cpp ../../dg_board/src/echoear/info.cpp ../../dg_board/test_host/touch_test.cpp)
target_include_directories(touch_driver_test PRIVATE ../../dg_board/test_host/fakes ../../dg_board/test_host/include ../../dg_board/include)

include(panel_trace.cmake)

foreach(board ECHOEAR WAVESHARE_146 P4XC)
    add_executable(canvas_${board} canvas_test.cpp)
    target_include_directories(canvas_${board} PRIVATE ../include ../../dg_board/include)
    if(board STREQUAL WAVESHARE_146)
        set(canvas 412)
    else()
        set(canvas 360)
    endif()
    target_compile_definitions(canvas_${board} PRIVATE CONFIG_DG_BOARD_${board}=1 DG_EXPECT_CANVAS=${canvas})
    target_link_libraries(canvas_${board} PRIVATE GTest::gtest_main)
    gtest_discover_tests(canvas_${board} TEST_PREFIX ${board}.)
endforeach()
