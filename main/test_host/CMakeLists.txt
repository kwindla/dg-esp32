cmake_minimum_required(VERSION 3.22)
project(app_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.18.0.tar.gz URL_HASH SHA256=6e3191c1455468b3fc35a417fb565c1c5071aee1b7e7f85e30cf48a98d37d8b5)
set(INSTALL_GTEST OFF)
set(BUILD_GMOCK OFF)
FetchContent_MakeAvailable(googletest)
add_executable(app_tests test_app.cpp test_liveness.cpp test_audio_slot.cpp test_app_wait.cpp test_demo_touch_delivery.cpp test_touch_init_retry.cpp test_rx_recording.cpp test_fixture_load.cpp ../app.cpp)
target_compile_definitions(app_tests PRIVATE DG_APP_HOST=1)
FetchContent_Declare(nlohmann_json
    URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
    URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(nlohmann_json)
target_link_libraries(app_tests PRIVATE nlohmann_json::nlohmann_json)
target_include_directories(app_tests PRIVATE .. ../../components/dg_audio/include ../../components/pipecat_esp_peer/src)
target_compile_options(app_tests PRIVATE -Wall -Wextra)
target_link_libraries(app_tests PRIVATE GTest::gtest_main)
enable_testing()
include(GoogleTest)
gtest_discover_tests(app_tests)

# Quoted includes resolve from the production source directory. Copy these
# unmodified inputs into the build tree so App/platform boundaries can be faked.
set(BOOT_TEST_SOURCE_DIR ${CMAKE_CURRENT_BINARY_DIR}/demo_boot_source)
file(MAKE_DIRECTORY ${BOOT_TEST_SOURCE_DIR})
foreach(input demo_boot.cpp demo_boot.h demo_settings.h demo_hold.h demo_intent.h demo_assets.h)
    configure_file(../${input} ${BOOT_TEST_SOURCE_DIR}/${input} COPYONLY)
endforeach()
add_executable(demo_boot_tests test_demo_boot.cpp ${BOOT_TEST_SOURCE_DIR}/demo_boot.cpp)
target_include_directories(demo_boot_tests PRIVATE ${BOOT_TEST_SOURCE_DIR} demo_boot_fakes)
target_link_libraries(demo_boot_tests PRIVATE GTest::gtest_main)
gtest_discover_tests(demo_boot_tests)

add_executable(heap_attribution_tests test_heap_attribution.cpp test_heap_new_callers.cpp ../heap_attribution.cpp ../heap_new_callers.cpp ../heap_new_wraps.cpp ../heap_tcp_state.cpp)
